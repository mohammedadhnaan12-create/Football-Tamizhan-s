// Automatic RSS Feed Fetcher
const rssUrl = 'https://api.rss2json.com/v1/api.json?rss_url=https%3A%2F%2Fwww.skysports.com%2Frss%2F12040';

fetch(rssUrl)
.then(response => response.json())
.then(data => {
const container = document.getElementById('rss-news-container');
container.innerHTML = '';

if(data.items && data.items.length > 0) {
data.items.slice(0, 6).forEach(item => {
const card = document.createElement('div');
card.className = 'card';

card.innerHTML = &lt;h3&gt;⚽ ${item.title}</h3>
<p>${item.description.replace(/&lt;[^&gt;]*&gt;?/gm, '').substring(0, 110)}...&lt;/p&gt; &lt;a href="${item.link}" target="_blank"&gt;Read More ➔&lt;/a&gt;;

container.appendChild(card);
});
} else {
container.innerHTML = '<p style="text-align:center; width:100%;">Unable to load news right now.</p>';
}
})
.catch(error => {
console.error('Error fetching RSS feed:', error);
document.getElementById('rss-news-container').innerHTML = '<p style="text-align:center; width:100%;">Error loading daily news updates.</p>';
});
</script>
