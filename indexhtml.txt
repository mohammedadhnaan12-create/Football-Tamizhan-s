// Robust RSS Feed Fetcher
const targetRss = encodeURIComponent('https://www.skysports.com/rss/12040');
const apiUrl = https://api.allorigins.win/get?url=${targetRss}`;

fetch(apiUrl)
.then(response => response.json())
.then(data => {
const parser = new DOMParser();
const xmlDoc = parser.parseFromString(data.contents, "text/xml");
const items = xmlDoc.querySelectorAll("item");

const container = document.getElementById('rss-news-container');
container.innerHTML = '';

if(items.length > 0) {
let count = 0;
items.forEach(item => {
if(count < 6) {
const title = item.querySelector("title") ? item.querySelector("title").textContent : "";
const link = item.querySelector("link") ? item.querySelector("link").textContent : "#";
const description = item.querySelector("description") ? item.querySelector("description").textContent.replace(/<[^>]*>?/gm, '').substring(0, 100) : "";

const card = document.createElement('div');
card.className = 'card';

card.innerHTML = &lt;h3&gt;⚽${title}</h3>
<p>{link}" target="_blank">Read More ➔</a>
`;

container.appendChild(card);
count++;
}
});
} else {
container.innerHTML = '<p style="text-align:center; width:100%;">No news available right now.</p>';
}
})
.catch(error => {
console.error('Error fetching RSS:', error);
document.getElementById('rss-news-container').innerHTML = '<p style="text-align:center; width:100%;">Daily news updating soon...</p>';
});
</script>
